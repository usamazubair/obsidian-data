
Add it after the build validation and before restarting PM2:

```
npm run build

if [ ! -d "$APP_DIR/dist" ]; then
    echo "ERROR: Build completed but the dist directory was not created."
    exit 1
fi

if [ ! -f "$APP_DIR/dist/main.js" ]; then
    echo "ERROR: dist/main.js was not created."
    exit 1
fi

# ---------------------------------------------------------
# 10. Clear Redis cache
# ---------------------------------------------------------

echo "Checking Redis connection..."

if ! redis-cli ping >/dev/null 2>&1; then
    echo "ERROR: Redis is not reachable."
    exit 1
fi

echo "Clearing all Redis databases..."

if ! redis-cli FLUSHALL; then
    echo "ERROR: Redis FLUSHALL failed."
    exit 1
fi

echo "Redis cache cleared successfully."

# ---------------------------------------------------------
# 11. Restart PM2
# ---------------------------------------------------------

echo "Checking PM2 application..."

if ! pm2 describe "$PM2_APP_NAME" >/dev/null 2>&1; then
    echo "ERROR: PM2 application '$PM2_APP_NAME' was not found for root."
    echo "Current root PM2 processes:"
    pm2 list
    exit 1
fi

echo "Restarting PM2 application: $PM2_APP_NAME..."

pm2 restart "$PM2_APP_NAME" --update-env

echo "Saving PM2 process list..."
pm2 save

echo "PM2 status:"
pm2 status "$PM2_APP_NAME"
```
